class RecordingRenderer implements CanvasRenderer {
  readonly subscriptions = new Map<RendererInputListener, RendererInputDispositionListener | undefined>();
  disposed = false;
  updateDocument(_update: RendererDocumentUpdate): void {}
  updateSession(_snapshot: SessionSnapshot): void {}
  updateInteraction(): void {}
  subscribeInput(listener: RendererInputListener, decide?: RendererInputDispositionListener): () => void {
    this.subscriptions.set(listener, decide); return () => { this.subscriptions.delete(listener); };
  }
  hitTest(point: ScreenPoint): HitResult { return { type: 'canvas', worldPoint: point }; }
  capturePointer(_id: number): void {}
  releasePointer(_id: number): void {}
  focus(): void {}
  dispose(): Promise<void> { this.disposed = true; this.subscriptions.clear(); return Promise.resolve(); }
  emit(input: RendererInput): { control: InputDisposition | undefined; behavior: InputDisposition | undefined } {
    let control: InputDisposition | undefined, behavior: InputDisposition | undefined;
    const entries = Array.from(this.subscriptions);
    for (const [, decide] of entries) control = decide?.(input) ?? control;
    for (const [listener] of entries) behavior = listener(input) ?? behavior;
    return { control, behavior };
  }
}
