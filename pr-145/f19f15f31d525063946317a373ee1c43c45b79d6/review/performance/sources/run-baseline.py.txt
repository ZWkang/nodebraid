from pathlib import Path
import datetime, hashlib, json, random, statistics, subprocess, sys, time

root = Path(__file__).resolve().parent
cases = []
for size in (1000, 10000, 50000):
    for operation in ('move-one', 'move-1pct', 'rewire-one', 'net-zero'):
        cases.append(dict(topology='sparse', size=size, operation=operation, retained=0))
for size in (10000, 50000):
    for operation in ('move-one', 'rewire-one'):
        cases.append(dict(topology='hub', size=size, operation=operation, retained=0))
    cases.append(dict(topology='nested', size=size, operation='move-one', retained=0))
    cases.append(dict(topology='sparse', size=size, operation='move-one', retained=20))

def case_id(case):
    return f"{case['topology']}-{case['size']}-{case['operation']}-retain{case['retained']}"

plan = dict(repetitions=3, warmupCount=20, sampleCount=30, scheduleSeed=137,
            processIsolation='new Bun process for every case/repetition',
            gc='natural; no forced collection', cases=cases)
(root/'matrix-plan.json').write_text(json.dumps(plan, indent=2)+'\n')
results_dir=root/'baseline'
results_dir.mkdir()
receipts=[]
all_results=[]
schedule=[]
for repeat in range(1, plan['repetitions']+1):
    order=cases.copy()
    random.Random(plan['scheduleSeed']+repeat).shuffle(order)
    schedule += [(repeat, case) for case in order]
(root/'schedule.json').write_text(json.dumps(schedule,indent=2)+'\n')
for step,(repeat,case) in enumerate(schedule,1):
    label=case_id(case)+f"-r{repeat}"
    output=results_dir/(label+'.json')
    cmd=['bun','--no-install',str(root/'benchmark.mjs'),case['topology'],str(case['size']),
         case['operation'],str(plan['warmupCount']),str(plan['sampleCount']),str(case['retained']),str(output)]
    started=datetime.datetime.now(datetime.timezone.utc).isoformat()
    tick=time.monotonic()
    with (results_dir/(label+'.log')).open('w') as log:
        result=subprocess.run(cmd,stdout=log,stderr=subprocess.STDOUT,cwd=root)
    receipt=dict(label=label,command=cmd,startedAt=started,exitCode=result.returncode,elapsedSeconds=time.monotonic()-tick)
    receipts.append(receipt)
    (root/'baseline-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
    if result.returncode:
        print(f"FAILED {label}; see log",flush=True)
        raise SystemExit(result.returncode)
    value=json.loads(output.read_text())
    assert value['checks']==plan['warmupCount']+plan['sampleCount']
    assert value['sampleCount']==len(value['timingsMs'])
    assert value['nodes']==case['size'] and value['edges']==2*case['size']
    all_results.append(dict(repetition=repeat,caseId=case_id(case),result=value))
    print(f"{step}/{len(schedule)} {label} median={value['statistics']['medianMs']:.3f}ms p95={value['statistics']['p95Ms']:.3f}ms",flush=True)

summary=[]
for case in cases:
    found=[r['result'] for r in all_results if r['caseId']==case_id(case)]
    assert len(found)==plan['repetitions']
    times=sorted(x for r in found for x in r['timingsMs'])
    medians=[r['statistics']['medianMs'] for r in found]
    summary.append(dict(**case,caseId=case_id(case),pooledMedianMs=statistics.median(times),
                        pooledP95Ms=times[__import__('math').ceil(len(times)*.95)-1],
                        pooledMaxMs=max(times),runMedianMinMs=min(medians),runMedianMaxMs=max(medians),
                        runMediansMs=medians,sampleCount=len(times),
                        rssBefore=[r['memoryBefore']['rss'] for r in found],
                        rssAfter=[r['memoryAfter']['rss'] for r in found]))
(root/'baseline-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
(root/'harness-sha256.txt').write_text(hashlib.sha256((root/'benchmark.mjs').read_bytes()).hexdigest()+'  benchmark.mjs\n')
print('BASELINE COMPLETE',flush=True)
