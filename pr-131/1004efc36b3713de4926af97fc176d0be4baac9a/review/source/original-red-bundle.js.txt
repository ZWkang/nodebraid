function copyAndFreezeDiagnosticValue(value, path, ancestors) {

    const firstPath = ancestors.get(value);
    if (firstPath) {
      throw new DiagnosticValueError(`Diagnostic value at ${path} must not be circular; first seen at ${firstPath}.`, {
        path,
        issue: "CIRCULAR",
        firstPath

      const symbolKey = Object.getOwnPropertySymbols(value)[0];
      if (symbolKey) {
        throw new DiagnosticValueError(`Diagnostic value at ${path} must use string keys; received ${String(symbolKey)}.`, { path, issue: "SYMBOL_KEY", receivedKey: String(symbolKey) });
      }
      if (Array.isArray(value)) {
        const copy = new Array(value.length);
        for (let index = 0;index < value.length; index += 1) {
          let descriptor;
          for (let owner = value;owner !== null; owner = Object.getPrototypeOf(owner)) {
            descriptor = Object.getOwnPropertyDescriptor(owner, index);
            if (descriptor)
              break;
          }
          if (!descriptor)
            continue;
          const propertyPath = `${path}[${index}]`;
          if (!("value" in descriptor)) {
            throw new DiagnosticValueError(`Diagnostic value at ${propertyPath} must use a data property; received accessor.`, { path: propertyPath, issue: "ACCESSOR_PROPERTY" });
          }
          copy[index] = copyAndFreezeDiagnosticValue(descriptor.value, propertyPath, ancestors);
        }
        return Object.freeze(copy);
      }
      const prototype = Object.getPrototypeOf(value);
      if (prototype !== Object.prototype && prototype !== null) {
        const constructorName = prototype?.constructor?.name || "object";
        throw new DiagnosticValueError(`Diagnostic value at ${path} must be a plain record; received ${constructorName}.`, { path, issue: "NON_PLAIN_OBJECT", receivedClass: constructorName });
      }
      return Object.freeze(Object.fromEntries(Object.keys(value).sort().map((key) => {
        const propertyPath = `${path}.${key}`;
        const descriptor = Object.getOwnPropertyDescriptor(value, key);
        if (!descriptor || !("value" in descriptor)) {
          throw new DiagnosticValueError(`Diagnostic value at ${propertyPath} must use a data property; received accessor.`, { path: propertyPath, issue: "ACCESSOR_PROPERTY" });
        }
        return [key, copyAndFreezeDiagnosticValue(descriptor.value, propertyPath, ancestors)];
      })));
    } finally {
      ancestors.delete(value);
    }
  }

    return value;

    if (!Number.isFinite(value)) {

  const provides = normalizeServiceBindings(providesInput, "provides");
  const requiredTokens = new Set(Object.values(requires));
  for (const token of Object.values(provides)) {
    if (requiredTokens.has(token)) {
      throw new PluginHostError("INVALID_DEFINITION", `Plugin "${name ?? "<anonymous>"}" cannot require and provide Service Token "${token.name}".`);
    }
  }
  const frozenDefinition = Object.freeze({
    name,
    requires,
    provides,
    setup
  });
  const plugin = Object.freeze({ name, requires, provides });
  internalPlugins.set(plugin, { definition: frozenDefinition });
  return plugin;
}
function getInternalPlugin(plugin) {
  return internalPlugins.get(plugin);

function normalizeServiceBindings(bindings, field) {
  if (bindings === undefined)
    return Object.freeze({});
  const inspection = inspectPlainRecord(bindings, "invalid-value");
  if (inspection.status === "invalid-value") {
    throw new PluginHostError("INVALID_DEFINITION", `Plugin ${field} must be a Service Binding record.`);
  }
  if (inspection.status === "invalid-prototype") {
    throw new PluginHostError("INVALID_DEFINITION", `Plugin ${field} must be a plain Service Binding record.`);
  }
  if (inspection.status === "invalid-key") {
    throw new PluginHostError("INVALID_DEFINITION", `Plugin ${field} must contain only enumerable string Service Bindings.`);
  }
  const normalized = {};
  for (const binding of inspection.keys) {
    const descriptor = Object.getOwnPropertyDescriptor(inspection.record, binding);
    if (descriptor === undefined || !descriptor.enumerable) {
      throw new PluginHostError("INVALID_DEFINITION", `Plugin ${field} must contain only enumerable string Service Bindings.`);
    }
    const token = Reflect.get(inspection.record, binding);
    getServiceName(token);
    Object.defineProperty(normalized, binding, {
      enumerable: true,
      value: token
    });
  }
  return Object.freeze(normalized);
}

class PluginGraph {
  #nodes = new Set;
  #providerReservations = new Map;
  register(pluginName, requires, provides) {
    const node = { pluginName, requires, provides };
    const providedTokens = [...new Set(Object.values(provides))];
    for (const token of providedTokens) {
      const existing = this.#providerReservations.get(token);
      if (existing) {
        throw new PluginHostError("PROVIDER_CONFLICT", `Plugin "${pluginName ?? "<anonymous>"}" cannot provide Service Token "${token.name}" because it is reserved by "${existing.pluginName ?? "<anonymous>"}".`, {
          type: "provider-conflict",
          serviceName: token.name,
          existingProvider: existing.pluginName ?? "<anonymous>",
          conflictingProvider: pluginName ?? "<anonymous>"
        });
      }
    }
    const candidateProviders = new Map(this.#providerReservations);
    for (const token of providedTokens)
      candidateProviders.set(token, node);
    const cycle = findDependencyCycle([...this.#nodes, node], candidateProviders);
    if (cycle) {
      throw new PluginHostError("DEPENDENCY_CYCLE", formatDependencyCycle(cycle), createDependencyCycleDetails(cycle));
    }
    this.#nodes.add(node);
    for (const token of providedTokens)
      this.#providerReservations.set(token, node);
    return node;
  }
  remove(node) {
    this.#nodes.delete(node);
    for (const token of new Set(Object.values(node.provides))) {
      if (this.#providerReservations.get(token) === node) {
        this.#providerReservations.delete(token);
      }
    }
  }
}
function findDependencyCycle(nodes, providers) {
  const visiting = new Set;
  const visited = new Set;
  const nodeStack = [];
  const edgeStack = [];
  const visit = (node) => {
    visiting.add(node);
    nodeStack.push(node);
    for (const token of Object.values(node.requires)) {
      const provider = providers.get(token);
      if (!provider)
        continue;
      const edge = { consumer: node, provider, token };
      if (visiting.has(provider)) {
        const cycleStart = nodeStack.indexOf(provider);
        return [...edgeStack.slice(cycleStart), edge];
      }
      if (visited.has(provider))
        continue;
      edgeStack.push(edge);
      const cycle = visit(provider);
      if (cycle)
        return cycle;
      edgeStack.pop();
    }
    nodeStack.pop();
    visiting.delete(node);
    visited.add(node);
    return;
  };
  for (const node of nodes) {
    if (visited.has(node))
      continue;
    const cycle = visit(node);
    if (cycle)
      return cycle;
  }
  return;
}
function formatDependencyCycle(cycle) {
  let path = `"${cycle[0].consumer.pluginName ?? "<anonymous>"}"`;
  for (const edge of cycle) {
    path += ` --[${edge.token.name}]--> "${edge.provider.pluginName ?? "<anonymous>"}"`;
  }
  return `Plugin dependency cycle: ${path}.`;
}
function createDependencyCycleDetails(cycle) {
  return Object.freeze({
    type: "dependency-cycle",
    path: Object.freeze(cycle.map((edge) => Object.freeze({
      plugin: edge.consumer.pluginName ?? "<anonymous>",
      serviceName: edge.token.name,
      provider: edge.provider.pluginName ?? "<anonymous>"
    })))
  });
}

function validateProvidedServices(pluginName, provides, result) {
  const bindingsByToken = new Map;
  for (const [binding, token] of Object.entries(provides)) {
    const duplicate = bindingsByToken.get(token);
    if (duplicate !== undefined) {
      throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" declares Provided Service Token "${token.name}" through both "${duplicate}" and "${binding}".`);
    }
    bindingsByToken.set(token, binding);
  }
  const declaredBindings = Object.keys(provides);
  if (!declaredBindings.length) {
    if (result !== undefined) {
      throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" returned a value without declaring a Provided Service.`);
    }
    return Object.freeze(Object.create(null));
  }
  const inspection = inspectPlainRecord(result, "invalid-prototype");
  if (inspection.status === "invalid-value") {
    throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" must return a Provided Service record.`);
  }
  if (inspection.status === "invalid-prototype") {
    throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" must return a plain Provided Service record.`);
  }
  if (inspection.status === "invalid-key") {
    throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" returned undeclared Provided Service binding "${String(inspection.key)}".`);
  }
  for (const binding of inspection.keys) {
    if (!Object.hasOwn(provides, binding)) {
      throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" returned undeclared Provided Service binding "${String(binding)}".`);
    }
  }
  const normalized = Object.create(null);
  for (const binding of declaredBindings) {
    if (!Object.hasOwn(inspection.record, binding)) {
      throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" did not return Provided Service binding "${binding}".`);
    }
    const value = inspection.record[binding];
    if (value == null) {
      throw new PluginHostError("CONTRACT_VIOLATION", `Plugin "${pluginName ?? "<anonymous>"}" returned ${String(value)} for Provided Service binding "${binding}".`);
    }
    normalized[binding] = value;
  }
  return Object.freeze(normalized);
}

class PluginActivation {
  definition;
  config;
  installChild;
  publishService;
  diagnostics;
  controller = new AbortController;
  resources = [];
  #disposal;
  #ended = false;
  constructor(definition, config, installChild, publishService, diagnostics) {
    this.definition = definition;
    this.config = config;
    this.installChild = installChild;
    this.publishService = publishService;
    this.diagnostics = diagnostics;
  }
  async run(context) {
    const services = Object.fromEntries(Object.entries(this.definition.requires ?? {}).map(([binding, token]) => [
      binding,
      context.get(getServiceName(token))
    ]));
    try {
      const result = await this.definition.setup({
        diagnostics: this.diagnostics,
        signal: this.controller.signal,
        services: Object.freeze(services),
        own: (dispose) => {
          this.#assertActive();
          this.resources.push(dispose);
        },
        install: (plugin, ...args) => {
          this.#assertActive();
          const child = this.installChild(plugin, ...args);
          this.resources.push(() => child.dispose());
          return child;
        }
      }, this.config);
      return validateProvidedServices(this.definition.name, this.definition.provides, result);
    } catch (error) {
      try {
        await this.dispose("setup-failed");
      } catch {}
      throw error;
    }
  }
  dispose(reason = "installation-disposed") {
    if (!this.#disposal) {
      this.#ended = true;
      this.controller.abort();
      this.#disposal = (async () => {
        try {
          const errors = [];
          for (const dispose of this.resources.splice(0).reverse()) {
            try {
              await dispose();
            } catch (error) {
              errors.push(error);
            }
          }
          if (errors.length) {
            throw new AggregateError(errors, "Plugin resource cleanup failed.");
          }
        } finally {
          this.diagnostics.emit({
            name: runtimeDiagnosticEvents.activationEnded,
            level: "debug",
            attributes: { reason }
          });
        }
      })();
    }
    return this.#disposal;
  }
  publishProvidedServices(services) {
    return Object.entries(this.definition.provides ?? {}).map(([binding, token]) => this.publishService(getServiceName(token), services[binding]));
  }
  #assertActive() {
    if (this.#ended) {
      throw new PluginHostError("INSTALLATION_DISPOSED", "Plugin Activation has ended.");
    }
  }
}

class CordisPluginInstallation {
  definition;
  config;
  diagnostics;
  getMissing;
  installChild;
  publishService;
  reportDependencyCleanupError;
  onDisposed;
  #snapshot;
  #listeners = new Set;
  #waiters = new Set;
  #fiber;
  #currentActivation;
  #currentDeactivation;
  #cleanupErrors = [];
  #disposal;
  #disposeRequested = false;
  #failed = false;
  #terminalError;
  constructor(definition, config, diagnostics, missing, getMissing, installChild, publishService, reportDependencyCleanupError, onDisposed) {
    this.definition = definition;
    this.config = config;
    this.diagnostics = diagnostics;
    this.getMissing = getMissing;
    this.installChild = installChild;
    this.publishService = publishService;
    this.reportDependencyCleanupError = reportDependencyCleanupError;
    this.onDisposed = onDisposed;
    this.#snapshot = createPendingSnapshot(missing);
    this.#emitStatusChanged("none", this.#snapshot);
  }
  attachFiber(fiber) {
    this.#fiber = fiber;
  }
  getSnapshot() {
    return this.#snapshot;
  }
  subscribe(listener) {
    this.#listeners.add(listener);
    return () => {
      this.#listeners.delete(listener);
    };
  }
  whenActive(signal) {
    if (this.#disposeRequested) {
      return Promise.reject(this.#disposedError());
    }
    if (this.#snapshot.status === "active")
      return Promise.resolve();
    if (this.#snapshot.status === "failed") {
      return Promise.reject(this.#snapshot.error);
    }
    if (signal?.aborted)
      return Promise.reject(getAbortReason(signal));
    return new Promise((resolve, reject) => {
      const waiter = {
        resolve,
        reject,
        signal,
        onAbort: signal ? () => {
          this.#waiters.delete(waiter);
          reject(getAbortReason(signal));
        } : undefined
      };
      if (waiter.onAbort)
        signal.addEventListener("abort", waiter.onAbort, { once: true });
      this.#waiters.add(waiter);
    });
  }
  async activate(context) {
    if (this.#failed)
      throw this.#terminalError;
    const activationDiagnostics = this.diagnostics.createActivation();
    activationDiagnostics.emit({
      name: runtimeDiagnosticEvents.activationStarted,
      level: "debug"
    });
    const activation = new PluginActivation(this.definition, this.config, this.installChild, this.publishService, activationDiagnostics);
    this.#currentActivation = activation;
    const requiredServiceNames = new Set(Object.values(this.definition.requires ?? {}).map(getServiceName));
    let disappearingServiceName;
    context.on("internal/service", (name) => {
      if (requiredServiceNames.has(name) && context.get(name) === undefined) {
        disappearingServiceName ??= name;
        activation.controller.abort();
      }
    });
    let providedServices;
    try {
      providedServices = await activation.run(context);
    } catch (error) {
      if (activation.controller.signal.aborted && (this.#disposeRequested || this.getMissing().length > 0) && isCancellationError(error, activation.controller.signal)) {
        try {
          await activation.dispose(this.#disposeRequested ? "installation-disposed" : "dependency-lost");
        } catch (cleanupError) {
          if (!this.#disposeRequested) {
            this.#failDependencyCleanup(activation, disappearingServiceName, cleanupError);
            return async () => {};
          }
        }
        if (!this.#disposeRequested) {
          this.#setSnapshot(createPendingSnapshot(this.getMissing()));
        }
        return async () => {};
      }
      this.#fail(error, "setup");
      throw error;
    }
    if (activation.controller.signal.aborted || this.#disposeRequested) {
      try {
        await activation.dispose(this.#disposeRequested ? "installation-disposed" : "dependency-lost");
      } catch (cleanupError) {
        if (!this.#disposeRequested) {
          this.#failDependencyCleanup(activation, disappearingServiceName, cleanupError);
          return async () => {};
        }
      }
      if (!this.#disposeRequested) {
        this.#setSnapshot(createPendingSnapshot(this.getMissing()));
      }
      return async () => {};
    }
    const withdrawals = activation.publishProvidedServices(providedServices);
    Promise.resolve().then(() => Promise.resolve(this.#fiber)).then(() => {
      if (this.#currentActivation === activation && !activation.controller.signal.aborted && !this.#disposeRequested && !this.#failed) {
        this.#setSnapshot(Object.freeze({ status: "active" }));
        this.#resolveWaiters();
      }
    }).catch((error) => {
      if (this.#disposeRequested || this.#failed)
        return;
      this.#fail(error, "activation");
    });
    let deactivation;
    const deactivate = () => {
      deactivation ??= (async () => {
        activation.controller.abort();
        const dependencyDriven = !this.#disposeRequested && (disappearingServiceName !== undefined || this.getMissing().length > 0);
        if (dependencyDriven && !this.#failed) {
          this.#setSnapshot(createPendingSnapshot(this.getMissing()));
        }
        const errors = [];
        for (const withdraw of withdrawals.reverse()) {
          try {
            await withdraw();
          } catch (error) {
            collectCleanupError(errors, error);
          }
        }
        try {
          await activation.dispose(this.#disposeRequested ? "installation-disposed" : dependencyDriven ? "dependency-lost" : "setup-completed");
        } catch (error) {
          collectCleanupError(errors, error);
        }
        if (this.#currentActivation === activation) {
          this.#currentActivation = undefined;
          this.#currentDeactivation = undefined;
        }
        if (!this.#disposeRequested && !this.#failed && !dependencyDriven) {
          this.#setSnapshot(createPendingSnapshot(this.getMissing()));
        }
        if (errors.length) {
          const error = new AggregateError(errors, "Plugin resource cleanup failed.");
          if (!this.#disposeRequested && dependencyDriven) {
            this.#failDependencyCleanup(activation, disappearingServiceName, error);
            return;
          }
          this.#cleanupErrors.push(...errors);
          if (!this.#disposeRequested) {
            this.#fail(error, "cleanup");
            throw error;
          }
        }
      })();
      return deactivation;
    };
    this.#currentDeactivation = deactivate;
    return deactivate;
  }
  dispose() {
    if (!this.#disposal) {
      let completeDisposal;
      let failDisposal;
      this.#disposal = new Promise((resolve, reject) => {
        completeDisposal = resolve;
        failDisposal = reject;
      });
      this.#disposeRequested = true;
      this.#rejectWaiters(this.#disposedError());
      this.#currentActivation?.controller.abort();
      const deactivation = this.#currentDeactivation?.();
      if (deactivation) {
        Promise.resolve(deactivation).catch(() => {});
      }
      Promise.resolve().then(() => this.#fiber.dispose()).then(async () => {
        try {
          await this.#currentActivation?.dispose("installation-disposed");
        } catch (error) {
          collectCleanupError(this.#cleanupErrors, error);
        }
        if (this.#cleanupErrors.length) {
          throw new AggregateError(this.#cleanupErrors, "Plugin resource cleanup failed.");
        }
      }).then(() => {
        this.onDisposed();
        this.#setSnapshot(Object.freeze({ status: "disposed" }));
      }, (error) => {
        this.onDisposed();
        this.#setSnapshot(Object.freeze({ status: "disposed" }));
        const diagnosticError = this.diagnostics.findUndiagnosedError(error);
        if (diagnosticError.found) {
          this.diagnostics.diagnostics.emit({
            name: runtimeDiagnosticEvents.installationDisposeFailed,
            level: "error",
            attributes: { phase: "cleanup" },
            error: diagnosticError.error
          });
        }
        throw error;
      }).then(completeDisposal, failDisposal);
    }
    return this.#disposal;
  }
  #setSnapshot(snapshot, failurePhase) {
    if (this.#snapshot === snapshot)
      return;
    const previousStatus = this.#snapshot.status;
    this.#snapshot = snapshot;
    this.#emitStatusChanged(previousStatus, snapshot, failurePhase);
    for (const listener of this.#listeners) {
      try {
        listener();
      } catch (error) {
        this.diagnostics.diagnostics.reportFault(error, {
          name: runtimeDiagnosticEvents.installationSubscriberFault,
          attributes: { status: snapshot.status }
        });
      }
    }
  }
  #emitStatusChanged(from, snapshot, failurePhase) {
    this.diagnostics.diagnostics.emit({
      name: runtimeDiagnosticEvents.installationStatusChanged,
      level: snapshot.status === "failed" ? "error" : "debug",
      attributes: {
        from,
        to: snapshot.status,
        ...snapshot.status === "pending" ? { missingServiceNames: snapshot.missing.map((service) => service.name).sort() } : {},
        ...snapshot.status === "failed" && failurePhase ? { phase: failurePhase } : {}
      },
      ...snapshot.status === "failed" ? { error: snapshot.error } : {}
    });
  }
  #resolveWaiters() {
    for (const waiter of this.#waiters) {
      if (waiter.onAbort)
        waiter.signal.removeEventListener("abort", waiter.onAbort);
      waiter.resolve();
    }
    this.#waiters.clear();
  }
  #rejectWaiters(error) {
    for (const waiter of this.#waiters) {
      if (waiter.onAbort)
        waiter.signal.removeEventListener("abort", waiter.onAbort);
      waiter.reject(error);
    }
    this.#waiters.clear();
  }
  #failDependencyCleanup(activation, disappearingServiceName, error) {
    const aggregate = error instanceof AggregateError ? error : new AggregateError([error], "Plugin resource cleanup failed.");
    if (this.#currentActivation === activation)
      this.#currentActivation = undefined;
    this.#fail(aggregate, "cleanup");
    const missingService = this.getMissing()[0];
    const serviceName = disappearingServiceName ?? (missingService ? getServiceName(missingService) : undefined);
    if (serviceName)
      this.reportDependencyCleanupError(serviceName, aggregate);
  }
  #fail(error, phase) {
    this.#failed = true;
    this.#terminalError = error;
    this.#setSnapshot(Object.freeze({ status: "failed", error }), phase);
    this.#rejectWaiters(error);
  }
  #disposedError() {
    return new PluginHostError("INSTALLATION_DISPOSED", "Plugin Installation has been disposed.");
  }
}
function createPendingSnapshot(missing) {
  return Object.freeze({
    status: "pending",
    missing: Object.freeze([...missing])
  });
}

function applyChangeSetToDraft(changeSet, direction, draftNodes, draftEdges, nodeWriter, edgeWriter) {
  const changes = validateChangeSet(changeSet, direction);
  for (const change of changes) {
    if (change.entity === "node") {
      const expected = direction === "forward" ? change.before : change.after;
      const actual = draftNodes.get(change.id) ?? null;
      if (!canvasNodesEqual(actual, expected))
        throwChangeSetConflict(change, direction, expected, actual);
    } else {
      const expected = direction === "forward" ? change.before : change.after;
      const actual = draftEdges.get(change.id) ?? null;
      if (!canvasEdgesEqual(actual, expected))
        throwChangeSetConflict(change, direction, expected, actual);
    }
  }
  const orderedChanges = direction === "forward" ? changes : [...changes].reverse();
  for (const change of orderedChanges) {
    const source = direction === "forward" ? change.before : change.after;
    const target = direction === "forward" ? change.after : change.before;
    const writer = change.entity === "node" ? nodeWriter : edgeWriter;
    if (source === null && target !== null) {
      writer.add(target);
    } else if (source !== null && target === null) {
      writer.remove(change.id);
    } else if (target !== null) {
      writer.replace(change.id, target);
    }
  }
}
function validateChangeSet(changeSet, direction) {
  if (direction !== "forward" && direction !== "reverse") {
    if (typeof direction !== "string") {
      throwInvalidChangeSet("Change Set direction must be forward or reverse.", {
        issue: "INVALID_DIRECTION",
        receivedType: direction === null ? "null" : typeof direction
      });
    }
    throwInvalidChangeSet("Change Set direction must be forward or reverse.", {
      issue: "INVALID_DIRECTION",
      direction
    });
  }
  if (!isRecord(changeSet)) {
    throwInvalidChangeSet("Change Set must be an object.", {});
  }
  const beforeRevision = changeSet.beforeRevision;
  const revision = changeSet.revision;
  if (!Number.isSafeInteger(beforeRevision) || beforeRevision < 0 || !Number.isSafeInteger(revision) || revision !== beforeRevision + 1) {
    throwInvalidChangeSet("Change Set revisions must be adjacent non-negative safe integers.", {
      issue: "INVALID_REVISIONS",
      beforeRevision: describeInvalidRevision(beforeRevision),
      revision: describeInvalidRevision(revision)
    });
  }
  const changes = changeSet.changes;
  if (!Array.isArray(changes)) {
    throwInvalidChangeSet("Change Set must contain at least one change.", {});
  }
  const length = changes.length;
  if (length === 0) {
    throwInvalidChangeSet("Change Set must contain at least one change.", {});
  }
  const identities = new Set;
  const accepted = [];
  for (const index of Array.prototype.keys.call({ length })) {
    const change = changes[index];
    if (!isRecord(change)) {
      throwInvalidChangeSet("Change must identify a Node or Edge.", { index });
    }
    const entityKind = change.entity;
    if (entityKind !== "node" && entityKind !== "edge") {
      throwInvalidChangeSet("Change must identify a Node or Edge.", { index });
    }
    const id = change.id;
    if (typeof id !== "string" || id.length === 0) {
      throwInvalidChangeSet("Change entity ID must be a non-empty string.", { index });
    }
    const identity = `${entityKind}\x00${id}`;
    if (identities.has(identity)) {
      throwInvalidChangeSet("Change Set must not contain duplicate entity changes.", {
        index,
        entity: entityKind,
        id
      });
    }
    identities.add(identity);
    const before = change.before;
    const after = change.after;
    if (before === null && after === null) {
      throwInvalidChangeSet("Change must contain a before or after entity.", { index });
    }
    const captureSide = (entity, side) => {
      if (entity === null)
        return null;
      const captured = entityKind === "node" ? captureCanvasNodeValue(entity) : captureCanvasEdgeValue(entity);
      if (!captured || captured.id !== id) {
        throwInvalidChangeSet("Change entity must match its kind and ID.", {
          index,
          side,
          entity: entityKind,
          id
        });
      }
      return captured;
    };
    const ownedBefore = captureSide(before, "before");
    const ownedAfter = captureSide(after, "after");
    accepted.push(Object.freeze({ entity: entityKind, id, before: ownedBefore, after: ownedAfter }));
  }
  return accepted;
}
function throwChangeSetConflict(change, direction, expected, actual) {
  throw new KernelError("CHANGE_SET_CONFLICT", `Change Set source does not match ${change.entity} "${change.id}".`, Object.freeze({
    entity: change.entity,
    id: change.id,
    direction,
    expectedState: expected === null ? "missing" : "present",
    actualState: actual === null ? "missing" : "present"
  }));
}
function throwInvalidChangeSet(message, details) {
  throw new KernelError("INVALID_CHANGE_SET", message, Object.freeze({ ...details }));
}
function describeInvalidRevision(value) {
  if (typeof value !== "number") {
    return Object.freeze({
      receivedType: value === null ? "null" : typeof value
    });
  }
  if (!Number.isFinite(value)) {
    return Object.freeze({ receivedNumber: describeNonFiniteNumber(value) });
  }
  return Object.freeze({ value });
}

class GraphIndex {
  #incoming = new Map;
  #outgoing = new Map;
  #children = new Map;
  constructor(nodes, edges) {
    for (const node of nodes.values()) {
      if (node.parentId !== undefined)
        append(this.#children, node.parentId, node);
    }
    for (const edge of edges.values()) {
      append(this.#outgoing, edge.source.nodeId, edge);
      append(this.#incoming, edge.target.nodeId, edge);
    }
    sortBuckets(this.#children);
    sortBuckets(this.#incoming);
    sortBuckets(this.#outgoing);
  }
  addNode(node) {
    if (node.parentId !== undefined)
      insertSorted(this.#children, node.parentId, node);
  }
  replaceNode(before, after) {
    if (before.parentId !== undefined)
      removeById(this.#children, before.parentId, before.id);
    if (after.parentId !== undefined)
      insertSorted(this.#children, after.parentId, after);
  }
  removeNode(node) {
    if (node.parentId !== undefined)
      removeById(this.#children, node.parentId, node.id);
  }
  addEdge(edge) {
    insertSorted(this.#outgoing, edge.source.nodeId, edge);
    insertSorted(this.#incoming, edge.target.nodeId, edge);
  }
  replaceEdge(before, after) {
    this.removeEdge(before);
    this.addEdge(after);
  }
  removeEdge(edge) {
    removeById(this.#outgoing, edge.source.nodeId, edge.id);
    removeById(this.#incoming, edge.target.nodeId, edge.id);
  }
  getIncoming(nodeId) {
    return frozenCopy(this.#incoming.get(nodeId));
  }
  getOutgoing(nodeId) {
    return frozenCopy(this.#outgoing.get(nodeId));
  }
  getIncident(nodeId) {
    return Object.freeze(mergeById(this.#incoming.get(nodeId) ?? [], this.#outgoing.get(nodeId) ?? []));
  }
  getChildren(parentId) {
    return frozenCopy(this.#children.get(parentId));
  }
}

      assertOpen();

function append(buckets, key, value) {
  const bucket = buckets.get(key);
  if (bucket === undefined)
    buckets.set(key, [value]);
  else
    bucket.push(value);
}
function sortBuckets(buckets) {
  for (const bucket of buckets.values())
    bucket.sort((left, right) => compareIds(left.id, right.id));
}
function insertSorted(buckets, key, value) {
  const bucket = buckets.get(key);
  if (bucket === undefined) {
    buckets.set(key, [value]);
    return;
  }
  let low = 0;
  let high = bucket.length;
  while (low < high) {
    const middle = low + high >>> 1;
    if (compareIds(bucket[middle].id, value.id) < 0)
      low = middle + 1;
    else
      high = middle;
  }
  bucket.splice(low, 0, value);
}
function removeById(buckets, key, id) {
  const bucket = buckets.get(key);
  if (bucket === undefined)
    return;
  const index = bucket.findIndex((value) => value.id === id);
  if (index >= 0)
    bucket.splice(index, 1);
  if (bucket.length === 0)
    buckets.delete(key);
}
function frozenCopy(values) {
  return Object.freeze(values === undefined ? [] : [...values]);
}
function mergeById(left, right) {
  const merged = [];
  let leftIndex = 0;
  let rightIndex = 0;
  while (leftIndex < left.length || rightIndex < right.length) {
    const leftValue = left[leftIndex];
    const rightValue = right[rightIndex];
    if (leftValue === undefined) {
      merged.push(rightValue);
      rightIndex += 1;
    } else if (rightValue === undefined) {
      merged.push(leftValue);
      leftIndex += 1;
    } else {
      const comparison = compareIds(leftValue.id, rightValue.id);
      if (comparison <= 0) {
        merged.push(leftValue);
        leftIndex += 1;
      }
      if (comparison >= 0) {
        if (comparison > 0)
          merged.push(rightValue);
        rightIndex += 1;
      }
    }
  }
  return merged;
}

function createEntityWriter(options) {
  const label = options.kind === "node" ? "Node" : "Edge";
  return Object.freeze({
    add(entity) {
      options.assertOpen();
      const id = entity.id;
      if (options.draft.has(id)) {
        throw new KernelError("ENTITY_ALREADY_EXISTS", `${label} "${id}" already exists.`, Object.freeze({ entity: options.kind, id }));
      }
      const frozen = options.freeze(entity, id);
      options.draft.set(id, frozen);
      options.onAdd(frozen);
      options.touchedIds.add(id);
    },
    replace(id, entity) {
      options.assertOpen();
      const before = options.draft.get(id);
      if (before === undefined) {
        throw new KernelError("ENTITY_NOT_FOUND", `${label} "${id}" does not exist.`, Object.freeze({ entity: options.kind, id }));
      }
      const actualId = entity.id;
      if (actualId !== id) {
        throw new KernelError("ENTITY_ID_MISMATCH", `Replacement ${label} ID "${actualId}" does not match "${id}".`, Object.freeze({ entity: options.kind, id, actualId }));
      }
      const frozen = options.freeze(entity, actualId);
      options.draft.set(id, frozen);
      options.onReplace(before, frozen);
      options.touchedIds.add(id);
    },
    remove(id) {
      options.assertOpen();
      const entity = options.draft.get(id);
      if (entity === undefined) {
        throw new KernelError("ENTITY_NOT_FOUND", `${label} "${id}" does not exist.`, Object.freeze({ entity: options.kind, id }));
      }
      options.draft.delete(id);
      options.onRemove(entity);
      options.touchedIds.add(id);
    }
  });
}

function validateGraph(nodes, edges) {
  const issues = [];
  for (const node of nodes.values()) {
    for (const coordinate of ["x", "y"]) {
      const value = node.position[coordinate];
      if (!Number.isFinite(value)) {
        issues.push(Object.freeze({ code: "INVALID_POSITION", nodeId: node.id, coordinate, value }));
      }
    }
    if (node.size !== undefined) {
      for (const dimension of ["width", "height"]) {
        const value = node.size[dimension];
        if (!Number.isFinite(value) || value < 0) {
          issues.push(Object.freeze({ code: "INVALID_SIZE", nodeId: node.id, dimension, value }));
        }
      }
    }
    if (node.parentId !== undefined && !nodes.has(node.parentId)) {
      issues.push(Object.freeze({ code: "MISSING_PARENT", nodeId: node.id, parentId: node.parentId }));
    }
  }
  for (const edge of edges.values()) {
    for (const endpoint of ["source", "target"]) {
      const nodeId = edge[endpoint].nodeId;
      if (!nodes.has(nodeId)) {
        issues.push(Object.freeze({ code: "MISSING_EDGE_ENDPOINT", edgeId: edge.id, endpoint, nodeId }));
      }
    }
  }
  issues.push(...findParentCycles(nodes));
  return issues.sort(compareGraphIssues);
}
function describeGraphIssues(issues) {
  return issues.map((issue) => {
    switch (issue.code) {
      case "MISSING_EDGE_ENDPOINT":
        return {
          code: issue.code,
          edgeId: issue.edgeId,
          endpoint: issue.endpoint,
          nodeId: issue.nodeId
        };
      case "MISSING_PARENT":
        return { code: issue.code, nodeId: issue.nodeId, parentId: issue.parentId };
      case "PARENT_CYCLE":
        return { code: issue.code, nodeIds: issue.nodeIds };
      case "INVALID_POSITION":
        return {
          code: issue.code,
          nodeId: issue.nodeId,
          coordinate: issue.coordinate,
          receivedNumber: describeNonFiniteNumber(issue.value)
        };
      case "INVALID_SIZE":
        return {
          code: issue.code,
          nodeId: issue.nodeId,
          dimension: issue.dimension,
          ...Number.isFinite(issue.value) ? { value: issue.value } : { receivedNumber: describeNonFiniteNumber(issue.value) }
        };
    }
  });
}
function findParentCycles(nodes) {
  const completed = new Set;
  const cycles = [];
  for (const startId of [...nodes.keys()].sort(compareIds)) {
    if (completed.has(startId))
      continue;
    const path = [];
    const pathIndexes = new Map;
    let currentId = startId;
    while (currentId !== undefined && nodes.has(currentId) && !completed.has(currentId)) {
      const cycleStart = pathIndexes.get(currentId);
      if (cycleStart !== undefined) {
        const nodeIds = Object.freeze(path.slice(cycleStart).sort(compareIds));
        cycles.push(Object.freeze({ code: "PARENT_CYCLE", nodeIds }));
        break;
      }
      pathIndexes.set(currentId, path.length);
      path.push(currentId);
      currentId = nodes.get(currentId)?.parentId;
    }
    for (const nodeId of path)
      completed.add(nodeId);
  }
  return cycles;
}
function compareGraphIssues(left, right) {
  const codeOrder = graphIssueOrder[left.code] - graphIssueOrder[right.code];
  if (codeOrder !== 0)
    return codeOrder;
  switch (left.code) {
    case "MISSING_EDGE_ENDPOINT":
      if (right.code !== left.code)
        return 0;
      return compareIds(left.edgeId, right.edgeId) || compareIds(left.endpoint, right.endpoint);
    case "MISSING_PARENT":
      if (right.code !== left.code)
        return 0;
      return compareIds(left.nodeId, right.nodeId);
    case "PARENT_CYCLE":
      if (right.code !== left.code)
        return 0;
      return compareIds(left.nodeIds.join("\x00"), right.nodeIds.join("\x00"));
    case "INVALID_POSITION":
      if (right.code !== left.code)
        return 0;
      return compareIds(left.nodeId, right.nodeId) || compareIds(left.coordinate, right.coordinate);
    case "INVALID_SIZE":
      if (right.code !== left.code)
        return 0;
      return compareIds(left.nodeId, right.nodeId) || sizeDimensionOrder[left.dimension] - sizeDimensionOrder[right.dimension];
  }
}
var graphIssueOrder = Object.freeze({
  MISSING_EDGE_ENDPOINT: 0,
  MISSING_PARENT: 1,
  PARENT_CYCLE: 2,
  INVALID_POSITION: 3,
  INVALID_SIZE: 4
});
var sizeDimensionOrder = Object.freeze({ width: 0, height: 1 });

        freeze: freezeCanvasEdge,
        assertOpen: assertTransactionOpen,
        onAdd: (edge) => draftIndex.addEdge(edge),
        onReplace: (oldEdge, newEdge) => draftIndex.replaceEdge(oldEdge, newEdge),
        onRemove: (edge) => draftIndex.removeEdge(edge)
      });
      const transaction = Object.freeze({
        query,
        nodes: nodeWriter,
        edges: edgeWriter,
        applyChangeSet(changeSet, direction) {
          assertTransactionOpen();
          applyChangeSetToDraft(changeSet, direction, draftNodes, draftEdges, nodeWriter, edgeWriter);
        }
      });
      transactionActive = true;
      try {
        let callbackResult;
        try {
          callbackResult = callback(transaction);
        } finally {
          transactionOpen = false;
        }
        if (isThenable(callbackResult)) {
          throw new KernelError("ASYNC_TRANSACTION", "Transaction callback must complete synchronously.", Object.freeze({}));
        }
        const issues = validateGraph(draftNodes, draftEdges);
        if (issues.length > 0) {
          throw new KernelError("INVALID_GRAPH", "The final graph is invalid.", Object.freeze({ issues: describeGraphIssues(issues) }));
        }
        const nodeChanges = collectEntityChanges("node", nodes, draftNodes, touchedNodeIds, canvasNodesEqual);
        const edgeChanges = collectEntityChanges("edge", edges, draftEdges, touchedEdgeIds, canvasEdgesEqual);
        const changes = Object.freeze([...nodeChanges, ...edgeChanges]);
        if (changes.length === 0)
          return null;
        const revision = incrementRevision(before.snapshot.revision);
        if (revision === null) {
          throw new KernelError("REVISION_OVERFLOW", "Kernel revision cannot exceed Number.MAX_SAFE_INTEGER.", Object.freeze({ revision: before.snapshot.revision }));
        }
        const after = createCanvasView(revision, draftNodes, draftEdges);
        const origin = metadata?.origin;
        const commandId = metadata?.commandId;
        const changeSet = Object.freeze({
          beforeRevision: before.snapshot.revision,
          revision,
          ...origin === undefined ? {} : { origin },
          ...commandId === undefined ? {} : { commandId },
          changes
        });
        const commit = Object.freeze({ before, after, changeSet });
        nodes = draftNodes;
        edges = draftEdges;
        view = after;
        return commit;
      } finally {
        transactionActive = false;
      }
    }
  });
}

    return false;
  return typeof Reflect.get(value, "then") === "function";
}

function createPortRuntime(diagnostics) {
  const entries = new Map;
  const listeners = new Set;
  let generation = 0;
  let disposed = false;
  const assertActive = () => {
    if (disposed)
      throw new PortError("SERVICE_DISPOSED", "Port Service has been disposed.");
  };
  const notify = () => {
    for (const listener of Array.from(listeners)) {
      try {
        listener();
      } catch (error) {
        diagnostics.reportFault(error, { name: portDiagnosticEvents.subscriberFault });
      }
    }
  };
  const service = Object.freeze({
    register(value) {
      assertActive();
      const contribution = acceptContribution(value);
      if (entries.has(contribution.id)) {
        throw new PortError("CONTRIBUTION_ALREADY_REGISTERED", "Port Contribution ID is already registered.", {
          contributionId: contribution.id
        });
      }
      const entry = { contribution, disposed: false };
      entries.set(contribution.id, entry);
      generation += 1;
      notify();
      return Object.freeze({
        dispose() {
          if (entry.disposed)
            return;
          entry.disposed = true;
          if (entries.get(contribution.id) !== entry)
            return;
          entries.delete(contribution.id);
          generation += 1;
          if (!disposed)
            notify();
        }
      });
    },
    snapshot(document) {
      assertActive();
      if (!isCanvasSnapshot(document)) {
        throw new PortError("INVALID_CONTRIBUTION", "Port Snapshot requires a committed Canvas Snapshot.");
      }
      const contributions = Array.from(entries.values(), (entry) => entry.contribution).sort((left, right) => compareIds2(left.id, right.id));
      const nodes = [];
      for (const node of document.nodes) {
        const ports = resolveNodePorts(node, contributions);
        if (ports.length > 0)
          nodes.push(Object.freeze({ nodeId: node.id, ports }));
      }
      return Object.freeze({
        documentRevision: document.revision,
        generation,
        nodes: Object.freeze(nodes)
      });
    },
    isConnectionTargetDefined(node, target) {
      assertActive();
      if (node.id !== target.nodeId)
        return false;
      const contributions = Array.from(entries.values(), (entry) => entry.contribution).sort((left, right) => compareIds2(left.id, right.id));
      const definitions = resolveNodePorts(node, contributions);
      if (target.portId === undefined)
        return definitions.length === 0;
      return definitions.find((definition) => definition.id === target.portId)?.role === target.role;
    },
    subscribe(listener) {
      assertActive();
      if (typeof listener !== "function") {
        throw new PortError("INVALID_CONTRIBUTION", "Port subscriber must be a function.");
      }
      listeners.add(listener);
      let active = true;
      return () => {
        if (!active)
          return;
        active = false;
        listeners.delete(listener);
      };
    }
  });
  return {
    service,
    dispose() {
      if (disposed)
        return;
      disposed = true;
      listeners.clear();
      entries.clear();
      generation += 1;
    }
  };
}
function acceptContribution(value) {
  if (typeof value !== "object" || value === null) {
    throw new PortError("INVALID_CONTRIBUTION", "Port Contribution is malformed.");
  }
  const id = value.id;
  if (typeof id !== "string" || id.length === 0) {
    throw new PortError("INVALID_CONTRIBUTION", "Port Contribution is malformed.");
  }
  const definePorts = value.definePorts;
  if (typeof definePorts !== "function") {
    throw new PortError("INVALID_CONTRIBUTION", "Port Contribution is malformed.");
  }
  for (const key of Reflect.ownKeys(value)) {
    if (key === "id" || key === "definePorts")
      continue;
    throw new PortError("INVALID_CONTRIBUTION", "Port Contribution contains an unknown field.");
  }
  return Object.freeze({ id, definePorts });
}
function resolveNodePorts(node, contributions) {
  const definitions = new Map;
  for (const contribution of contributions) {
    const received = contribution.definePorts(node);
    if (!Array.isArray(received)) {
      throw new PortError("INVALID_PORT_DEFINITION", "Port Contribution must return an array.", {
        contributionId: contribution.id
      });
    }
    for (const value of received) {
      const definition = acceptPortDefinition(value, contribution.id);
      if (definitions.has(definition.id)) {
        throw new PortError("PORT_ALREADY_DEFINED", "Port ID is already defined for this Node.", {
          contributionId: contribution.id,
          portId: definition.id
        });
      }
      definitions.set(definition.id, definition);
    }
  }
  return Object.freeze(Array.from(definitions.values()).sort((left, right) => compareIds2(left.id, right.id)));
}
function acceptPortDefinition(value, contributionId) {
  if (typeof value !== "object" || value === null) {
    throw new PortError("INVALID_PORT_DEFINITION", "Port Definition is malformed.", { contributionId });
  }
  const id = Reflect.get(value, "id");
  if (typeof id !== "string" || id.length === 0) {
    throw new PortError("INVALID_PORT_DEFINITION", "Port Definition is malformed.", { contributionId });
  }
  const role = Reflect.get(value, "role");
  if (role !== "source" && role !== "target") {
    throw new PortError("INVALID_PORT_DEFINITION", "Port Definition is malformed.", { contributionId });
  }
  for (const key of Reflect.ownKeys(value)) {
    if (key === "id" || key === "role")
      continue;
    throw new PortError("INVALID_PORT_DEFINITION", "Port Definition contains an unknown field.", {
      contributionId
    });
  }
  return Object.freeze({ id, role });
}

      });

            }
          });

            throw new RendererError("INVALID_INPUT_SUBSCRIBER", "Renderer Input listener must be a function.", {
              receivedType: describeReceivedType2(listener)

              return;

          assertOperational();

          assertOperational();

        },
        releasePointer(pointerId) {
          assertOperational();

      }
    }

  };

  }

  }
  if (resolved.minZoom > resolved.maxZoom) {

  }

      }
      seenNodeIds.add(move.nodeId);
      if (previousNodeId !== undefined && previousNodeId > move.nodeId) {
        throw new InteractionError("INVALID_MOVE", "Move Nodes Command Node IDs must use canonical order.");
      }
      previousNodeId = move.nodeId;
      if (!Number.isFinite(move.basePosition.x) || !Number.isFinite(move.basePosition.y) || !Number.isFinite(move.position.x) || !Number.isFinite(move.position.y)) {
        throw new InteractionError("INVALID_MOVE", "Move Nodes Command positions must be finite.");
      }
    }
    return kernel.transact((transaction) => {
      for (const move of moves) {
        const node = transaction.query.getNode(move.nodeId);
        if (!node || node.position.x !== move.basePosition.x || node.position.y !== move.basePosition.y) {
          throw new InteractionError("STALE_GESTURE", "Move Nodes Command source position is stale.");
        }
      }
      for (const move of moves) {
        const node = transaction.query.getNode(move.nodeId);
        if (!node)
          throw new InteractionError("STALE_GESTURE", "Move Nodes Command target disappeared.");
        transaction.nodes.replace(move.nodeId, { ...node, position: move.position });
      }
    }, { origin: "interaction", commandId: execution.commandId });
  };
}
function captureMoveNodesInput(value) {
  if (!isRecord2(value))
    throwInvalidMoveInput();
  const moves = value.moves;
  if (!Array.isArray(moves))
    throwInvalidMoveInput();
  const length = +moves.length;
  const captured = [];
  for (const index of Array.prototype.keys.call({ length })) {
    const move = moves[index];
    if (!isRecord2(move))
      throwInvalidMoveInput();
    const id = move.nodeId;
    if (typeof id !== "string" || id.length === 0)
      throwInvalidMoveInput();
    const basePosition = captureMovePoint(move.basePosition);
    const position = captureMovePoint(move.position);
    captured.push({ nodeId: nodeId(id), basePosition, position });
  }
  return captured;
}
function captureMovePoint(value) {
  if (!isRecord2(value))
    throwInvalidMoveInput();
  const x = value.x;
  if (typeof x !== "number")
    throwInvalidMoveInput();
  const y = value.y;
  if (typeof y !== "number")
    throwInvalidMoveInput();
  return { x, y };
}
function throwInvalidMoveInput() {
  throw new InteractionError("INVALID_MOVE", "Move Nodes Command input is malformed.");
}

function defineInteractionShortcut(shortcut) {
  if (typeof shortcut !== "object" || shortcut === null || typeof shortcut.code !== "string" || shortcut.code.length === 0 || !Array.isArray(shortcut.modifierAlternatives) || shortcut.modifierAlternatives.length === 0) {
    throw new InteractionError("INVALID_SHORTCUT", "Interaction Keyboard Shortcut is malformed.");
  }
  const alternatives = shortcut.modifierAlternatives.map((value) => {
    if (typeof value !== "object" || value === null || typeof value.alt !== "boolean" || typeof value.control !== "boolean" || typeof value.meta !== "boolean" || typeof value.shift !== "boolean") {
      throw new InteractionError("INVALID_SHORTCUT", "Shortcut modifiers are malformed.");
    }
    return Object.freeze({ alt: value.alt, control: value.control, meta: value.meta, shift: value.shift });
  });
  return Object.freeze({ code: shortcut.code, modifierAlternatives: Object.freeze(alternatives) });
}

var preventDefaultDisposition = Object.freeze({ defaultAction: "prevent" });
function createShortcutRuntime(options) {
  const entries = new Set;
  const entriesByKey = new Map;
  let disposed = false;
  const disposeEntry = (entry) => {
    if (entry.disposePromise)
      return entry.disposePromise;
    let completeDisposal;
    entry.disposePromise = new Promise((resolve) => {
      completeDisposal = resolve;
    });
    for (const execution of entry.executions)
      execution.controller.abort();
    Promise.all([...entry.executions].map((execution) => execution.settled)).then(() => {
      for (const key of entry.keys) {
        if (entriesByKey.get(key) === entry)
          entriesByKey.delete(key);
      }
      entries.delete(entry);
      completeDisposal();
    });
    return entry.disposePromise;
  };
  const service = Object.freeze({
    register(binding) {
      if (disposed)
        throw new InteractionError("SHORTCUT_DISPOSED", "Interaction Shortcut Service is disposed.");
      const accepted = acceptBinding(binding);
      const keys = new Set(accepted.shortcut.modifierAlternatives.map((modifiers) => bindingKey(accepted.shortcut.code, modifiers)));
      for (const key of keys) {
        if (entriesByKey.has(key)) {
          throw new InteractionError("SHORTCUT_ALREADY_REGISTERED", "Interaction Shortcut Binding is already registered.");
        }
      }
      const entry = {
        binding: accepted,
        keys,
        executions: new Set
      };
      entries.add(entry);
      for (const key of keys)
        entriesByKey.set(key, entry);
      return Object.freeze({ dispose: () => disposeEntry(entry) });
    }
  });
  const handleInput = (input) => {
    if (disposed || input.type !== "key.down")
      return;
    const entry = entriesByKey.get(bindingKey(input.code, input.modifiers));
    if (!entry)
      return;
    if (entry.disposePromise)
      return preventDefaultDisposition;
    const gestureType = options.getActiveGestureType();
    if (input.repeat || gestureType) {
      options.diagnostics.emit({
        name: interactionDiagnosticEvents.inputRejected,
        level: "info",
        attributes: {
          inputType: input.type,
          ...gestureType ? { gestureType } : {},
          reason: input.repeat ? "repeat" : "active-gesture"
        }
      });
      return preventDefaultDisposition;
    }
    execute(entry);
    return preventDefaultDisposition;
  };
  const decideDisposition = (input) => {
    if (disposed || input.type !== "key.down")
      return;
    return entriesByKey.has(bindingKey(input.code, input.modifiers)) ? preventDefaultDisposition : undefined;
  };
  const execute = (entry) => {
    const controller = new AbortController;
    let complete;
    const settled = new Promise((resolve) => {
      complete = resolve;
    });
    const running = { controller, settled };
    entry.executions.add(running);
    options.commands.execute(entry.binding.command, undefined, { signal: controller.signal }).then(() => finish(), (error) => {
      const expectedDisposalAbort = entry.disposePromise !== undefined && controller.signal.aborted && error === controller.signal.reason;
      if (!expectedDisposalAbort) {
        if (error instanceof CommandError && error.code === "COMMAND_NOT_FOUND") {
          options.diagnostics.emit({
            name: interactionDiagnosticEvents.inputRejected,
            level: "info",
            attributes: { inputType: "key.down", reason: "capability-unavailable" }
          });
        } else {
          options.diagnostics.reportFault(error, {
            name: interactionDiagnosticEvents.commandFault,
            attributes: { commandId: entry.binding.command.id }
          });
        }
      }
      finish();
    });
    function finish() {
      entry.executions.delete(running);
      complete();
    }
  };
  return {
    service,
    decideDisposition,
    handleInput,
    async dispose() {
      disposed = true;
      await Promise.all([...entries].map(disposeEntry));
    }
  };
}
function acceptBinding(binding) {
  if (typeof binding !== "object" || binding === null || typeof binding.command !== "object" || binding.command === null) {
    throw new InteractionError("INVALID_SHORTCUT", "Interaction Shortcut Binding is malformed.");
  }
  const shortcut = defineInteractionShortcut(binding.shortcut);
  return Object.freeze({ shortcut, command: binding.command });
}
function bindingKey(code, modifiers) {
  return `${code}\x00${Number(modifiers.alt)}${Number(modifiers.control)}${Number(modifiers.meta)}${Number(modifiers.shift)}`;
}

function activateInteractionRuntime(context, config) {
  const renderer = context.services.renderer;
  const session = context.services.session;
  const kernel = context.services.kernel;
  const commands = context.services.commands;
  const ports = context.services.ports;
  let closing = false;
  let activeGesture;
  let spacePressed = false;
  const rejectedPointerIds = new Set;
  let moveRegistration;
  let createEdgeRegistration;
  let deleteSelectionRegistration;
  let stopKernel = () => {
    return;
  };
  let stopSession = () => {
    return;
  };
  let stopPorts = () => {
    return;
  };
  let stopInput = () => {
    return;
  };
  const projection = renderer.bindInteractionProjection();
  const shortcutRuntime = createShortcutRuntime({
    commands,
    diagnostics: context.diagnostics,
    getActiveGestureType: () => activeGesture?.type
  });
  context.own(() => shortcutRuntime.dispose());
  const reapplyGestureProjection = (gesture) => {
    if (gesture.type === "node-drag" && gesture.active && gesture.moves) {
      projection.update({ type: "node-drag", nodes: gesture.moves });
    } else if (gesture.type === "viewport-pan" && gesture.active && gesture.viewport) {
      projection.update({
        type: "viewport-pan",
        baseViewport: gesture.baseViewport,
        viewport: gesture.viewport
      });
    } else if (gesture.type === "connection") {
      projection.update({
        type: "connection-preview",
        source: gesture.source,
        pointerWorldPoint: gesture.pointerWorldPoint,
        target: gesture.target
      });
    } else if (gesture.type === "box-selection" && gesture.active && gesture.rect) {
      projection.update({ type: "box-selection", rect: gesture.rect });
    }
  };
  const cancelGesture = (gesture, reason, releasePointer) => {
    if (activeGesture !== gesture)
      return;
    activeGesture = undefined;
    const errors = [];
    const attempt = (operation) => {
      try {
        operation();
      } catch (error) {
        errors.push(error);
      }
    };
    if (hasPreview(gesture))
      attempt(() => projection.update(null));
    if (releasePointer)
      attempt(() => renderer.releasePointer(gesture.pointerId));
    attempt(() => emitGestureCancellation(context.diagnostics, gesture.type, reason));
    if (errors.length === 1)
      throw errors[0];
    if (errors.length > 1)
      throw new AggregateError(errors, "Interaction Gesture cancellation failed.");
  };
  context.signal.addEventListener("abort", () => {
    closing = true;
  });
  context.own(async () => {
    closing = true;
    const gesture = activeGesture;
    const cleanupErrors = [];
    const attempt = (cleanup) => {
      try {
        cleanup();
      } catch (error) {
        cleanupErrors.push(error);
      }
    };
    attempt(stopInput);
    attempt(stopSession);
    attempt(stopPorts);
    attempt(stopKernel);
    attempt(() => projection.dispose());
    if (gesture)
      attempt(() => renderer.releasePointer(gesture.pointerId));
    activeGesture = undefined;
    spacePressed = false;
    rejectedPointerIds.clear();
    if (gesture) {
      attempt(() => emitGestureCancellation(context.diagnostics, gesture.type, "lifecycle"));
    }
    if (moveRegistration) {
      try {
        await moveRegistration.dispose();
      } catch (error) {
        cleanupErrors.push(error);
      }
    }
    if (createEdgeRegistration) {
      try {
        await createEdgeRegistration.dispose();
      } catch (error) {
        cleanupErrors.push(error);
      }
    }
    if (deleteSelectionRegistration) {
      try {
        await deleteSelectionRegistration.dispose();
      } catch (error) {
        cleanupErrors.push(error);
      }
    }
    if (cleanupErrors.length > 0) {
      throw new AggregateError(cleanupErrors, "Interaction Runtime cleanup failed.");
    }
  });
  moveRegistration = commands.register(moveNodesCommand, createMoveNodesHandler(kernel));
  createEdgeRegistration = commands.register(createEdgeCommand, createEdgeHandler(kernel, ports));
  deleteSelectionRegistration = commands.register(deleteSelectionCommand, createDeleteSelectionHandler(kernel));
  stopKernel = kernel.observeCommits((commit) => {
    if (closing || !activeGesture)
      return;
    const gesture = activeGesture;
    if (gesture.type === "node-drag") {
      const stale = gesture.nodes.some((evidence) => {
        const node = commit.after.query.getNode(evidence.nodeId);
        return !node || node.position.x !== evidence.basePosition.x || node.position.y !== evidence.basePosition.y;
      });
      if (stale)
        cancelGesture(gesture, "stale", true);
      else
        reapplyGestureProjection(gesture);
    } else if (gesture.type === "viewport-pan") {
      reapplyGestureProjection(gesture);
    } else if (gesture.type === "connection") {
      if (!connectionAnchorAvailable(ports, gesture.source, commit.after.query.getNode(gesture.source.nodeId))) {
        cancelGesture(gesture, "stale", true);
      } else {
        if (gesture.target.type !== "none" && !connectionAnchorAvailable(ports, gesture.target.anchor, commit.after.query.getNode(gesture.target.anchor.nodeId))) {
          gesture.target = { type: "none" };
        }
        reapplyGestureProjection(gesture);
      }
    } else if (gesture.type === "box-selection") {
      reapplyGestureProjection(gesture);
    }
  });
  stopPorts = ports.subscribe(() => {
    if (closing || !activeGesture)
      return;
    const gesture = activeGesture;
    if (gesture.type === "connection") {
      const view = kernel.read();
      if (!connectionAnchorAvailable(ports, gesture.source, view.query.getNode(gesture.source.nodeId))) {
        cancelGesture(gesture, "stale", true);
        return;
      }
      if (gesture.target.type !== "none" && !connectionAnchorAvailable(ports, gesture.target.anchor, view.query.getNode(gesture.target.anchor.nodeId))) {
        gesture.target = { type: "none" };
      }
    }
    reapplyGestureProjection(gesture);
  });
  stopSession = session.subscribe(() => {
    if (closing || !activeGesture)
      return;
    const gesture = activeGesture;
    if (gesture.type === "viewport-pan" || gesture.type === "box-selection") {
      if (!viewportsEqual2(session.getSnapshot().viewport, gesture.baseViewport)) {
        cancelGesture(gesture, "stale", true);
      } else {
        reapplyGestureProjection(gesture);
      }
    } else if (gesture.type === "node-drag" || gesture.type === "connection") {
      reapplyGestureProjection(gesture);
    }
  });
  stopInput = renderer.subscribeInput((input) => {
    if (closing)
      return;
    if (input.type === "key.down" || input.type === "key.up") {
      const shortcutDisposition = shortcutRuntime.handleInput(input);
      if (shortcutDisposition)
        return shortcutDisposition;
      if (input.type === "key.down" && (input.key === "Delete" || input.code === "Delete")) {
        if (activeGesture) {
          context.diagnostics.emit({
            name: interactionDiagnosticEvents.inputRejected,
            level: "info",
            attributes: {
              inputType: "key.down",
              gestureType: activeGesture.type,
              reason: "active-gesture"
            }
          });
          return;
        }
        const selection = session.getSnapshot().selection;
        if (selection.nodeIds.length === 0 && selection.edgeIds.length === 0)
          return;
        executeDeleteSelection(commands, context.diagnostics, { selection, evidence: kernel.read().snapshot }, () => closing);
        return;
      }
      if (input.code === "Space")
        spacePressed = input.type === "key.down";
      if (input.type === "key.down" && input.code === "Escape" && (activeGesture?.type === "connection" || activeGesture?.type === "box-selection")) {
        cancelGesture(activeGesture, "escape", true);
      }
      return;
    }
    if (input.type === "focus.lost") {
      spacePressed = false;
      return;
    }
    if (input.type === "focus.gained")
      return;
    if (input.type === "wheel") {
      if (activeGesture) {
        context.diagnostics.emit({
          name: interactionDiagnosticEvents.inputRejected,
          level: "info",
          attributes: {
            inputType: "wheel",
            gestureType: activeGesture.type,
            reason: "active-gesture"
          }
        });
        return;
      }
      const current = session.getSnapshot().viewport;
      const zoom = Math.max(config.minZoom, Math.min(config.maxZoom, current.zoom * Math.exp(-input.deltaY * config.wheelZoomSensitivity)));
      if (zoom === current.zoom)
        return;
      session.setViewport({
        x: input.screenPoint.x - input.worldPoint.x * zoom,
        y: input.screenPoint.y - input.worldPoint.y * zoom,
        zoom
      });
      return;
    }
    if (input.type === "pointer.move" && rejectedPointerIds.has(input.pointerId))
      return;
    if ((input.type === "pointer.up" || input.type === "pointer.cancel") && rejectedPointerIds.delete(input.pointerId)) {
      return;
    }
    if (input.type === "pointer.down") {
      if (activeGesture) {
        if (input.pointerId !== activeGesture.pointerId && !rejectedPointerIds.has(input.pointerId)) {
          rejectedPointerIds.add(input.pointerId);
          context.diagnostics.emit({
            name: interactionDiagnosticEvents.pointerRejected,
            level: "info",
            attributes: {
              inputType: "pointer.down",
              gestureType: activeGesture.type,
              reason: "additional-pointer"
            }
          });
        }
        return;
      }
      if (input.button !== "primary" && input.button !== "auxiliary")
        return;
      const hit = renderer.hitTest(input.screenPoint);
      if (hit === null)
        return;
      const additive = input.modifiers.shift || input.modifiers.meta || input.modifiers.control;
      const connectionHit = toConnectionAnchor(hit);
      if (connectionHit?.role === "source" && config.connection && !additive && input.pointerType !== "mouse") {
        context.diagnostics.emit({
          name: interactionDiagnosticEvents.inputRejected,
          level: "info",
          attributes: { inputType: "pointer.down", gestureType: "connection", reason: "unsupported-pointer" }
        });
        return;
      }
      renderer.focus();
      renderer.capturePointer(input.pointerId);
      const selectionGesture = { type: "selection", pointerId: input.pointerId };
      activeGesture = selectionGesture;
      try {
        if (input.button === "auxiliary" || spacePressed) {
          activeGesture = {
            type: "viewport-pan",
            pointerId: input.pointerId,
            startScreenPoint: input.screenPoint,
            baseViewport: session.getSnapshot().viewport,
            active: false
          };
          return;
        }
        if (connectionHit?.role === "source" && config.connection && !additive && input.button === "primary") {
          const gesture = {
            type: "connection",
            pointerId: input.pointerId,
            source: connectionHit,
            pointerWorldPoint: input.worldPoint,
            target: { type: "none" }
          };
          activeGesture = gesture;
          reapplyGestureProjection(gesture);
          return;
        }
        if (hit.type === "canvas") {
          activeGesture = {
            type: "box-selection",
            pointerId: input.pointerId,
            startScreenPoint: input.screenPoint,
            startWorldPoint: input.worldPoint,
            baseViewport: session.getSnapshot().viewport,
            additive,
            active: false,
            completeClick: () => session.setSelection(computeClickSelection(session.getSnapshot().selection, hit, additive))
          };
          return;
        }
        if (additive) {
          selectionGesture.completeClick = () => session.setSelection(computeClickSelection(session.getSnapshot().selection, hit, true));
          return;
        }
        if (hit.type === "node") {
          const selection = session.getSnapshot().selection;
          const nodeIds = selection.nodeIds.includes(hit.nodeId) ? selection.nodeIds : [hit.nodeId];
          const view = kernel.read();
          activeGesture = {
            type: "node-drag",
            pointerId: input.pointerId,
            startScreenPoint: input.screenPoint,
            startWorldPoint: input.worldPoint,
            nodes: nodeIds.map((nodeId) => {
              const node = view.query.getNode(nodeId);
              if (!node)
                throw new Error("Node Drag candidate must exist in the current Canvas View.");
              return { nodeId, basePosition: node.position };
            }),
            active: false
          };
        }
        if (hit.type === "node" || hit.type === "port") {
          const selection = session.getSnapshot().selection;
          if (selection.nodeIds.includes(hit.nodeId) && (selection.nodeIds.length > 1 || selection.edgeIds.length > 0)) {
            const completeClick = () => session.setSelection(computeClickSelection(session.getSnapshot().selection, hit, false));
            if (activeGesture.type === "node-drag" || activeGesture.type === "selection") {
              activeGesture.completeClick = completeClick;
            }
            return;
          }
        }
        session.setSelection(computeClickSelection(session.getSnapshot().selection, hit, false));
      } catch (error) {
        activeGesture = undefined;
        try {
          renderer.releasePointer(input.pointerId);
        } catch (cleanupError) {
          throw new AggregateError([error, cleanupError], "Interaction Input handling and cleanup both failed.");
        }
        throw error;
      }
      return;
    }
    const gesture = activeGesture;
    if (input.type === "pointer.move" && gesture && input.pointerId === gesture.pointerId) {
      if (gesture.type === "node-drag") {
        if (!gesture.active && isBelowDragThreshold(gesture.startScreenPoint, input.screenPoint, config.dragThreshold)) {
          return;
        }
        gesture.active = true;
        gesture.completeClick = undefined;
        const worldDeltaX = input.worldPoint.x - gesture.startWorldPoint.x;
        const worldDeltaY = input.worldPoint.y - gesture.startWorldPoint.y;
        gesture.moves = gesture.nodes.map((node) => ({
          nodeId: node.nodeId,
          basePosition: node.basePosition,
          position: {
            x: node.basePosition.x + worldDeltaX,
            y: node.basePosition.y + worldDeltaY
          }
        }));
        reapplyGestureProjection(gesture);
      } else if (gesture.type === "viewport-pan") {
        const deltaX = input.screenPoint.x - gesture.startScreenPoint.x;
        const deltaY = input.screenPoint.y - gesture.startScreenPoint.y;
        if (!gesture.active && isBelowDragThreshold(gesture.startScreenPoint, input.screenPoint, config.dragThreshold)) {
          return;
        }
        gesture.active = true;
        gesture.viewport = {
          x: gesture.baseViewport.x + deltaX,
          y: gesture.baseViewport.y + deltaY,
          zoom: gesture.baseViewport.zoom
        };
        reapplyGestureProjection(gesture);
      } else if (gesture.type === "connection") {
        updateConnectionGesture(gesture, input.worldPoint, renderer.hitTest(input.screenPoint));
        reapplyGestureProjection(gesture);
      } else if (gesture.type === "box-selection") {
        if (!gesture.active && isBelowDragThreshold(gesture.startScreenPoint, input.screenPoint, config.dragThreshold)) {
          return;
        }
        const rect = createWorldRect(gesture.startWorldPoint, input.worldPoint);
        gesture.active = true;
        gesture.rect = rect;
        reapplyGestureProjection(gesture);
      }
      return;
    }
    if (input.type === "pointer.up" && gesture && input.pointerId === gesture.pointerId) {
      if (gesture.type === "connection") {
        activeGesture = undefined;
        const terminalErrors = [];
        try {
          updateConnectionGesture(gesture, input.worldPoint, renderer.hitTest(input.screenPoint));
        } catch (error) {
          terminalErrors.push(error);
        }
        for (const cleanup of [() => projection.update(null), () => renderer.releasePointer(gesture.pointerId)]) {
          try {
            cleanup();
          } catch (error) {
            terminalErrors.push(error);
          }
        }
        if (terminalErrors.length === 1)
          throw terminalErrors[0];
        if (terminalErrors.length > 1)
          throw new AggregateError(terminalErrors, "Connection Gesture terminal failed.");
        if (gesture.target.type === "valid" && config.connection) {
          executeCreateEdge(commands, context.diagnostics, config.connection.materializeEdge, gesture, () => closing);
        } else {
          emitGestureCancellation(context.diagnostics, "connection", "invalid-target");
        }
        return;
      }
      activeGesture = undefined;
      if (gesture.type === "node-drag" && gesture.active) {
        projection.update(null);
        if (gesture.moves)
          executeMoveNodes(commands, context.diagnostics, gesture.moves, () => closing);
      } else if (gesture.type === "viewport-pan" && gesture.active) {
        projection.update(null);
        if (gesture.viewport)
          session.setViewport(gesture.viewport);
      } else if (gesture.type === "box-selection" && gesture.active) {
        projection.update(null);
        const rect = createWorldRect(gesture.startWorldPoint, input.worldPoint);
        session.setSelection(computeBoxSelection(session.getSnapshot().selection, kernel.read().snapshot.nodes, rect, gesture.additive));
      } else if (gesture.type !== "viewport-pan") {
        gesture.completeClick?.();
      }
    } else if (input.type === "pointer.cancel" && gesture && input.pointerId === gesture.pointerId) {
      cancelGesture(gesture, "pointer-cancel", true);
    }
  }, (input) => input.type === "key.down" || input.type === "key.up" ? shortcutRuntime.decideDisposition(input) : undefined);
  return shortcutRuntime.service;
}
function hasPreview(gesture) {
  return gesture.type === "connection" || gesture.type !== "selection" && gesture.active;
}
function isBelowDragThreshold(start, current, threshold) {
  return Math.hypot(current.x - start.x, current.y - start.y) < threshold;
}
function updateConnectionGesture(gesture, pointerWorldPoint, hit) {
  gesture.pointerWorldPoint = pointerWorldPoint;
  if (!isConnectionAnchorHit(hit)) {
    gesture.target = { type: "none" };
    return;
  }
  if (hit.role !== "target") {
    gesture.target = { type: "none" };
    return;
  }
  const anchor = toConnectionAnchor(hit);
  if (!anchor) {
    gesture.target = { type: "none" };
    return;
  }
  gesture.target = hit.nodeId !== gesture.source.nodeId ? { type: "valid", anchor } : { type: "invalid", anchor };
}

var historyService = defineService("history");
var historyPlugin = definePlugin({
  name: "@nodebraid/plugin-history",
  requires: { commands: commandService, kernel: kernelService },
  provides: { history: historyService },
  setup(context) {
    const commands = context.services.commands;
    const kernel = context.services.kernel;
    const undoEntries = [];
    const redoEntries = [];
    const listeners = new Set;
    let snapshot = createSnapshot(false, false);
    let pendingOperation;
    let observedRevision = kernel.read().snapshot.revision;
    let disposed = false;
    let unsubscribe;
    let undoRegistration;
    let redoRegistration;
    let baselineRegistration;
    const completeCommittedOperationAfterClose = () => {
      const operation = pendingOperation;
      if (!disposed || operation?.state.phase !== "committed")
        return;
      pendingOperation = undefined;
      operation.complete(operation.state.expectedCommit);
    };
    const close = () => {
      if (disposed)
        return;
      disposed = true;
      listeners.clear();
      completeCommittedOperationAfterClose();
    };
    const releaseState = () => {
      undoEntries.length = 0;
      redoEntries.length = 0;
      pendingOperation = undefined;
    };
    context.signal.addEventListener("abort", close, { once: true });
    context.own(async () => {
      context.signal.removeEventListener("abort", close);
      close();
      const errors = [];
      try {
        unsubscribe?.();
      } catch (error) {
        errors.push(error);
      }
      for (const registration of [baselineRegistration, redoRegistration, undoRegistration]) {
        try {
          await registration?.dispose();
        } catch (error) {
          errors.push(error);
        }
      }
      releaseState();
      if (errors.length > 0) {
        throw new AggregateError(errors, "History resource cleanup failed.");
      }
    });
    const assertActive = () => {
      if (disposed) {
        throw new HistoryError("SERVICE_DISPOSED", "History Service Activation has been disposed.");
      }
    };
    const publishSnapshot = () => {
      const canUndo = undoEntries.length > 0;
      const canRedo = redoEntries.length > 0;
      if (snapshot.canUndo === canUndo && snapshot.canRedo === canRedo)
        return;
      snapshot = createSnapshot(canUndo, canRedo);
      for (const listener of Array.from(listeners)) {
        try {
          listener();
        } catch (error) {
          context.diagnostics.reportFault(error, {
            name: historyDiagnosticEvents.subscriberFault,
            attributes: { canUndo: snapshot.canUndo, canRedo: snapshot.canRedo }
          });
        }
      }
    };
    unsubscribe = kernel.observeCommits((commit) => {
      if (disposed)
        return;
      if (commit.changeSet.beforeRevision !== observedRevision) {
        throw new Error(`History expected Commit after revision ${observedRevision}, received ${commit.changeSet.beforeRevision}.`);
      }
      observedRevision = commit.changeSet.revision;
      const operation = pendingOperation;
      const operationState = operation?.state;
      const matchesOperation = operation !== undefined && operationState !== undefined && (operationState.phase === "transacting" || operationState.phase === "committed" && operationState.expectedCommit === commit);
      if (matchesOperation) {
        operation.state = operationState.phase === "committed" ? { phase: "observed", expectedCommit: operationState.expectedCommit, observedCommit: commit } : { phase: "observed-before-return", observedCommit: commit };
        if (operation.kind === "baseline") {
          undoEntries.length = 0;
          redoEntries.length = 0;
        } else if (operation.direction === "undo") {
          undoEntries.pop();
          redoEntries.push(operation.entry);
        } else {
          redoEntries.pop();
          undoEntries.push(operation.entry);
        }
        queueMicrotask(() => {
          if (operation.state.phase !== "observed" || operation.state.expectedCommit !== operation.state.observedCommit) {
            if (pendingOperation === operation)
              pendingOperation = undefined;
            operation.fail(new Error("History observed a different Commit while executing its Command."));
            return;
          }
          if (pendingOperation === operation)
            pendingOperation = undefined;
          operation.complete(commit);
        });
      } else {
        undoEntries.push(commit.changeSet);
        redoEntries.length = 0;
      }
      if (observedRevision === kernel.read().snapshot.revision)
        publishSnapshot();
    });
    const assertCanExecute = (commandId, signal) => {
      assertActive();
      signal.throwIfAborted();
      if (pendingOperation) {
        throw new HistoryError("HISTORY_BUSY", "Another History Command is still in progress.", Object.freeze({ activeCommandId: pendingOperation.commandId, requestedCommandId: commandId }));
      }
      const kernelRevision = kernel.read().snapshot.revision;
      if (observedRevision !== kernelRevision) {
        throw new HistoryError("HISTORY_NOT_CAUGHT_UP", "History has not observed the current Kernel revision.", Object.freeze({ observedRevision, kernelRevision }));
      }
    };
    const acceptCommittedOperation = (pending, commit) => {
      if (pending.state.phase === "transacting") {
        pending.state = { phase: "committed", expectedCommit: commit };
      } else if (pending.state.phase === "observed-before-return") {
        pending.state = {
          phase: "observed",
          expectedCommit: commit,
          observedCommit: pending.state.observedCommit
        };
      } else {
        throw new Error(`History Command entered unexpected phase "${pending.state.phase}" before Transaction return.`);
      }
      completeCommittedOperationAfterClose();
    };
    const replay = (direction, commandId, signal) => {
      assertCanExecute(commandId, signal);
      const entries = direction === "undo" ? undoEntries : redoEntries;
      const entry = entries.at(-1);
      if (!entry) {
        if (direction === "redo") {
          throw new HistoryError("REDO_EMPTY", "No History Entry is available to redo.");
        }
        throw new HistoryError("UNDO_EMPTY", "No History Entry is available to undo.");
      }
      const pending = {
        ...createPendingCommit(commandId),
        kind: "replay",
        direction,
        entry
      };
      pendingOperation = pending;
      try {
        const commit = kernel.transact((transaction) => transaction.applyChangeSet(entry, direction === "undo" ? "reverse" : "forward"), { origin: "history", commandId });
        if (!commit)
          throw new Error(`${direction} replay did not produce a Canvas Commit.`);
        acceptCommittedOperation(pending, commit);
        return pending.completion;
      } catch (error) {
        if (pendingOperation === pending)
          pendingOperation = undefined;
        throw error;
      }
    };
    undoRegistration = commands.register(undoCommand, (_input, execution) => {
      return replay("undo", execution.commandId, execution.signal);
    });
    redoRegistration = commands.register(redoCommand, (_input, execution) => {
      return replay("redo", execution.commandId, execution.signal);
    });
    baselineRegistration = commands.register(establishHistoryBaselineCommand, (update, execution) => {
      assertCanExecute(execution.commandId, execution.signal);
      const pending = {
        ...createPendingCommit(execution.commandId),
        kind: "baseline"
      };
      pendingOperation = pending;
      try {
        const commit = kernel.transact(update, { origin: "history", commandId: execution.commandId });
        if (commit) {
          acceptCommittedOperation(pending, commit);
        } else {
          undoEntries.length = 0;
          redoEntries.length = 0;
          pending.state = { phase: "unchanged" };
          if (!disposed)
            publishSnapshot();
          queueMicrotask(() => {
            if (pendingOperation === pending)
              pendingOperation = undefined;
            pending.complete(null);
          });
        }
        return pending.completion;
      } catch (error) {
        if (pendingOperation === pending)
          pendingOperation = undefined;
        throw error;
      }
    });
    const service = Object.freeze({
      getSnapshot() {
        assertActive();
        return snapshot;
      },
      subscribe(listener) {
        assertActive();
        listeners.add(listener);
        return () => listeners.delete(listener);
      }
    });
    return { history: service };
  }
});

}
